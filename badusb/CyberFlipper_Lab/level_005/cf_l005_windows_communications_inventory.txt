REM CyberFlipper Level 005 - Windows Communications Inventory
REM Safe: visible local inventory of communication and meeting tools only.
REM Output: Desktop report opened in Notepad.
DELAY 1000
GUI r
DELAY 500
STRING powershell -NoProfile
ENTER
DELAY 1200
STRING $r="$env:USERPROFILE\Desktop\cyberflipper_l005_communications_inventory.txt"; "CYBERFLIPPER LEVEL 005 - WINDOWS COMMUNICATIONS INVENTORY" | Out-File $r; Get-Date | Out-File $r -Append; $names="Outlook|Gmail|Slack|Teams|Zoom|Thunderbird|Webex|Discord|WhatsApp|Skype"; Get-ItemProperty HKLM:\Software\Microsoft\Windows\CurrentVersion\Uninstall\*,HKLM:\Software\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\* -ErrorAction SilentlyContinue | Where-Object {$_.DisplayName -match $names} | Select DisplayName,DisplayVersion,Publisher,InstallDate | Sort DisplayName | Format-Table -Auto | Out-File $r -Append; notepad $r
ENTER
